cmake_minimum_required(VERSION 3.15)
project(pciutils_static C)

# Automatically generate an empty config.h file inside the lib/ directory 
# to satisfy the #include "config.h" directives in the code.
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/lib/config.h")
    file(WRITE "${CMAKE_CURRENT_SOURCE_DIR}/lib/config.h" "/* Generated by CMake */\n")
endif()

# 1. Base files common to the library cross-platform
set(LIBPCI_SOURCES
    lib/init.c
    lib/access.c
    lib/generic.c
    lib/dump.c
    lib/names.c
    lib/filter.c
    lib/names-hash.c
    lib/names-parse.c
    lib/names-cache.c
    lib/params.c
    lib/caps.c
    lib/names-hwdb.c    # Resolves: pci_id_hwdb_free and pci_id_hwdb_lookup
    lib/emulated.c      # Resolves: pci_emulated_read
    lib/names-net.c     # <--- ADD THIS LINE HERE to resolve pci_id_net_lookup
)


# 2. Add all Windows-specific interface methods
if(WIN32)
    list(APPEND LIBPCI_SOURCES 
        lib/win32-cfgmgr32.c
        lib/win32-kldbg.c
        lib/win32-sysdbg.c
        lib/win32-helpers.c   # Resolves: win32_strerror, win32_enable_privilege, etc.
        lib/i386-ports.c
    )
endif()

# 3. Create the Static Library (.lib)
add_library(pci STATIC ${LIBPCI_SOURCES})

# 4. Critical Preprocessor definitions for native Windows execution
target_compile_definitions(pci PRIVATE 
    -DPCI_CONFIG_H
    -DPCI_OS_WINDOWS
    -DPCI_HAVE_64BIT_ADDRESS
    -DPCI_HAVE_PM_WIN32_CFGMGR32
    -DPCI_HAVE_PM_WIN32_KLDBG
    -DPCI_HAVE_PM_WIN32_SYSDBG
    -DPCI_HAVE_PM_INTEL_CONF1
    
    # Injected paths for offline identification databases
    -DPCI_PATH_IDS_DIR=""
    -DPCI_IDS="pci.ids"
)

# 5. Link Windows system hardware management library
if(WIN32)
    # Includes advapi32 for secure process token privilege modifications
    target_link_libraries(pci PRIVATE cfgmgr32 advapi32) 
endif()

# 6. Expose the internal directory for headers
target_include_directories(pci PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/lib)
